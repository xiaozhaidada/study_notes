## 结构化剪枝

### 1. 阶段

```
PyTorch 模型 -----> 结构化减枝 -----> onnx 模型 
```

### 2. 安装引用环境

```python
# 安装
pip install torch-pruning
# 引用
import torch-pruning
```

### 3. 结构化剪枝

```python
import torch
import torch_pruning as tp
from torchvision.models import resnet50

# -------------------------
# 1. 加载模型
# -------------------------

model = resnet50(weights=None)

state_dict = torch.load(
    "resnet50.pth",
    map_location="cpu"
)

model.load_state_dict(state_dict)
model.eval()


# -------------------------
# 2. 示例输入
# -------------------------

example_inputs = torch.randn(
    1, 3, 224, 224
)


# -------------------------
# 3. 定义重要性指标
# -------------------------

importance = tp.importance.MagnitudeImportance(
    p=2
)


# -------------------------
# 4. 不希望剪掉的层
# -------------------------

ignored_layers = []

# 最后的分类层通常不要剪输出类别数
for module in model.modules():
    if isinstance(module, torch.nn.Linear):
        if module.out_features == 1000:
            ignored_layers.append(module)


# -------------------------
# 5. 创建结构化剪枝器
# -------------------------

pruner = tp.pruner.MagnitudePruner(  // 结构化维度剪枝
    model,
    example_inputs,
    importance=importance,
    pruning_ratio=0.2,
    ignored_layers=ignored_layers,
)


# -------------------------
# 6. 执行剪枝
# -------------------------

pruner.step()


# -------------------------
# 7. 测试模型
# -------------------------

output = model(example_inputs)

print("output shape:", output.shape)


# -------------------------
# 8. 保存
# -------------------------

torch.save(
    model,
    "resnet50_pruned.pt"
)

# -------------------------
# 9. 导出 onnx 文件
# -------------------------
dummy_input = torch.randn(
    1, 3, 224, 224
)

torch.onnx.export(
    model,
    dummy_input,
    "resnet50_pruned.onnx",
    
    input_names=["input"],
    output_names=["output"],
    
    opset_version=13
)
```
